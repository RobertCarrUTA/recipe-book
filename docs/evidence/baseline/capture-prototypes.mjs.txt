import fs from 'node:fs/promises';
import http from 'node:http';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import * as playwright from 'playwright';
import { findBrowserExecutable } from '../../scripts/browser-executable.mjs';
const root = fileURLToPath(new URL('../..', import.meta.url));
const out = fileURLToPath(new URL('.', import.meta.url));
const allowed = new Set(['docs/design/concepts.html','docs/design/concepts.css','docs/design/concepts.js','data/recipes.json','js/recipe_formatting.js','js/normalization.js','js/normalization_rules.js']);
const mime = { '.html':'text/html', '.css':'text/css', '.js':'text/javascript', '.json':'application/json' };
const server = http.createServer(async(req,res)=>{
  const file = new URL(req.url,'http://localhost').pathname.slice(1);
  if (!allowed.has(file)) {res.writeHead(404);res.end();return;}
  const bytes = await fs.readFile(path.join(root,file));
  res.writeHead(200,{'content-type':`${mime[path.extname(file)]}; charset=utf-8`});res.end(bytes);
});
await new Promise(resolve=>server.listen(0,'127.0.0.1',resolve));
const origin = `http://127.0.0.1:${server.address().port}`;
const browser = await playwright.chromium.launch({executablePath:await findBrowserExecutable({playwright}),headless:true});
const report = { artifact:'Design prototypes; not the implemented frontend', browser:browser.version(),capturedAt:new Date().toISOString(),screenshots:[],messages:[] };
try {
  for(const concept of ['notebook','workbench']) for(const [label,width,height] of [['desktop',1440,900],['mobile',390,844]]) {
    const context=await browser.newContext({viewport:{width,height}});const page=await context.newPage();
    page.on('pageerror',error=>report.messages.push({concept,label,error:String(error)}));
    await page.goto(`${origin}/docs/design/concepts.html?concept=${concept}`,{waitUntil:'networkidle'});
    await page.waitForSelector('#recipes .recipe');
    for(const theme of concept==='notebook'&&label==='desktop'?['light','dark']:['light']) {
      if(theme==='dark') await page.locator('#theme').click();
      const file=`prototype-${concept}-${label}-${theme}.png`;
      await page.screenshot({path:path.join(out,file),animations:'disabled'});
      const layout=await page.evaluate(()=>({width:innerWidth,documentWidth:document.documentElement.scrollWidth,overflow:Math.max(0,document.documentElement.scrollWidth-innerWidth)}));
      report.screenshots.push({file,concept,label,theme,viewport:{width,height},...layout});
    }
    await context.close();
  }
}finally{await fs.writeFile(path.join(out,'prototype-report.json'),JSON.stringify(report,null,2)+'\n');await browser.close();await new Promise(resolve=>server.close(resolve));}
console.log(JSON.stringify(report));
