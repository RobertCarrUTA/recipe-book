const root = fileURLToPath(new URL('../..', import.meta.url));
const out = fileURLToPath(new URL('.', import.meta.url));
const expectedCommit = '3117d47b9cdd3a844d063fbde2ca6cd5ff3a83d2';
const commit = execFileSync('git', ['rev-parse', 'HEAD'], { cwd: root, encoding: 'utf8' }).trim();
assert.equal(commit, expectedCommit, 'Baseline evidence must correspond to the audited baseline commit');
const allowed = new Set([...(await getStaticShellAssetPaths()), '404.html', 'sw.js', 'data/recipes.json']);
const mime = { '.html': 'text/html', '.css': 'text/css', '.js': 'text/javascript', '.json': 'application/json', '.svg': 'image/svg+xml', '.webmanifest': 'application/manifest+json' };
const recipes = JSON.parse(await fs.readFile(path.join(root, 'data/recipes.json'), 'utf8'));
const target = recipes.find(r => r.title === 'Dutch Oven Chicken Pot Pie') || recipes[0];

const supplement = process.argv.includes('--mobile-completion');
const initialReport = {
  commit, capturedAt: new Date().toISOString(), browser: null, theme: 'existing dark only',

Search/open metrics are browser event-to-next-paint lab latency, not field INP. LCP/CLS sampled after initial app readiness plus 500 ms. Screenshots are viewport captures unless content is indicated.',

const report = supplement ? JSON.parse(await fs.readFile(path.join(out, 'baseline-report.json'), 'utf8')) : initialReport;
let origin;
const server = http.createServer(async (req, res) => {
  try {
    const u = new URL(req.url, origin);
    let p = decodeURIComponent(u.pathname);
    if (p === '/recipe-book') p = '/recipe-book/';
    if (p.startsWith('/recipe-book/')) p = p.slice('/recipe-book'.length);
    p = p === '/' ? 'index.html' : p.replace(/^\//, '');
    const navigation = /text\/html/.test(req.headers.accept || '');
    const fallback = !allowed.has(p) && navigation && /^[a-z0-9]+(?:-[a-z0-9]+)*$/.test(p);
    if (!allowed.has(p) && !fallback) { res.writeHead(404); res.end('Not found'); return; }
    const file = fallback ? '404.html' : p;
    const bytes = await fs.readFile(path.join(root, file));
    res.writeHead(fallback ? 404 : 200, { 'content-type': `${mime[path.extname(file)] || 'application/octet-stream'}; charset=utf-8`, 'cache-control': 'no-cache' });
    res.end(bytes);
  } catch { res.writeHead(500); res.end('Server error'); }
});
await new Promise(resolve => server.listen(0, '127.0.0.1', resolve));
origin = `http://127.0.0.1:${server.address().port}`;
const executablePath = await findBrowserExecutable({ playwright });
assert.ok(executablePath, 'A browser executable is required');
const browser = await playwright.chromium.launch({ executablePath, headless: true });

async function persist() {
  await fs.writeFile(path.join(out, 'baseline-report.json'), JSON.stringify(report, null, 2) + '\n');
  const md = [ '# Original application baseline', '', `Commit: \`${commit}\`. Browser: Chromium ${report.browser.version}. Theme: dark only.`, '', report.methodology, '', '## Screenshots', '', '| Viewport | Workflow | Image |', '| --- | --- | --- |', ...report.screenshots.map(s => `| ${s.viewport.width} × ${s.viewport.height} | ${s.state} | [${s.file}](${s.file}) |`), '', 'See baseline-report.json for raw measurements, errors, failed checks, and exact fixture metadata.', '' ];
  await fs.writeFile(path.join(out, 'README.md'), md.join('\n'));
}
function monitor(page, label) {
  page.on('console', msg => { if (['warning', 'error'].includes(msg.type())) report.messages.push({ label, kind: msg.type(), text: msg.text() }); });
  page.on('pageerror', error => report.messages.push({ label, kind: 'pageerror', text: String(error) }));
  page.on('requestfailed', req => report.messages.push({ label, kind: 'requestfailed', url: req.url().replace(origin, ''), text: req.failure()?.errorText }));
  page.on('response', res => { if (res.status() >= 400) report.messages.push({ label, kind: 'http', status: res.status(), url: res.url().replace(origin, '') }); });
}
async function open(page, pathname = '/') {
  await page.goto(`${origin}${pathname}${pathname.includes('?') ? '&' : '?'}debug=1`, { waitUntil: 'networkidle' });
  await page.waitForFunction(() => window.recipeBookDebug?.getState().recipes.length > 0);
}
async function screenshot(page, label, state, selector) {
  const file = `${label}-${state}.png`;
  if (selector) await page.locator(selector).screenshot({ path: path.join(out, file), animations: 'disabled' });
  else await page.screenshot({ path: path.join(out, file), animations: 'disabled' });
  report.screenshots.push({ file, label, state, viewport: page.viewportSize(), theme: 'dark', fullPage: false, selector: selector || null, url: page.url().replace(origin, ''), capturedAt: new Date().toISOString() });
}
async function measureLayout(page, label, state) {
  const dimensions = await page.evaluate(() => ({
    viewportWidth: innerWidth, documentWidth: document.documentElement.scrollWidth, bodyWidth: document.body.scrollWidth,
    overflowPx: Math.max(0, document.documentElement.scrollWidth - innerWidth),
    overflowingControls: [...document.querySelectorAll('button,input,select,textarea')].filter(el => {
      if (!el.getClientRects().length || el.closest('[hidden]')) return false;
      const r = el.getBoundingClientRect(); return r.width > 0 && (r.right > innerWidth + 1 || r.left < -1);
    }).map(el => ({ id: el.id, text: (el.getAttribute('aria-label') || el.textContent || '').trim().slice(0, 100) })).slice(0, 20),
  }));
  report.layouts.push({ label, state, viewport: page.viewportSize(), ...dimensions });
}
async function chooseRecipe(page) {
  await page.fill('#recipeSearch', target.title);
  await page.waitForFunction(({id,count}) => document.querySelectorAll('.recipe').length === count && document.querySelector(`.recipe[data-recipe-id="${id}"]`), { id: target.id, count: expectedSearchCount });
  const recipe = page.locator(`.recipe[data-recipe-id="${target.id}"]`);
  if ((await recipe.locator('.accordion-header').getAttribute('aria-expanded')) !== 'true') await recipe.locator('.accordion-header').click();
  return recipe;
}
async function changeView(page, view) {
  if (page.viewportSize().width < 980) await page.locator(`.mobile-view-tab[data-view="${view}"]`).click();
  if (view === 'grocery') await page.locator('#groceryPanel').scrollIntoViewIfNeeded();
}
async function runCase(name, fn) {
  try { await fn(); report.checks.push({ name, passed: true }); console.log(`ok ${name}`); }
  catch (error) { report.failures.push({ name, error: String(error), stack: error.stack }); console.error(`failed ${name}: ${error}`); }
  await persist();
}
